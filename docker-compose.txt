version: "2"

services:
  backend:
    build: ./backend
    image: lizochek-neko/backend:2.0        # п. 4.1 — имя и версия как у репозитория
    volumes:
      - ./data.txt:/app/data.txt          # п. 4.2 — данные вне контейнера
    networks:
      - internal
    restart: always                       # п. 4.5 — автоперезапуск после ребута ноды

  frontend:
    build: ./frontend
    image: lizochek-neko/frontend:2.0       # п. 4.1
    depends_on:
      - backend                           # п. 4.4 — зависимость от backend
    networks:
      - internal
    ports:
      - "8080:80"                         # только frontend смотрит наружу
    restart: always

networks:
  internal:                               # п. 4.3 — общая сеть без проброса наружу
    driver: bridge
